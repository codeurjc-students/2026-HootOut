DO $$
BEGIN

if NOT EXISTS  (
SELECT  schema_name
FROM	information_schema.schemata
WHERE	schema_name = 'HootOut' )
then 
	CREATE SCHEMA "HootOut";
end if;

end $$;

--CHAT MESSAGE ATTACHMENT TABLE--------------------------------------------------------------
----------------------------------------------------------------------------------

DO $$
BEGIN
if NOT EXISTS (SELECT * FROM INFORMATION_SCHEMA.TABLES
				WHERE TABLE_SCHEMA = 'HootOut'
				AND   TABLE_NAME = 'ChatMessageAttachment')
then
	CREATE TABLE "HootOut"."ChatMessageAttachment"(
		"Id" UUID NOT NULL PRIMARY KEY ,
		"MessageId" UUID NOT NULL,
		"MediaFileId" UUID NOT NULL,
		"Deleted" boolean NOT NULL DEFAULT false,
		"CreatedAt" timestamptz  NOT NULL,
		"ModifiedAt" timestamptz NOT NULL,
		CONSTRAINT fk_chat_message_attachment_message FOREIGN KEY ("MessageId") REFERENCES "HootOut"."ChatMessage" ("Id"),
		CONSTRAINT fk_chat_message_attachment_media_file FOREIGN KEY ("MediaFileId") REFERENCES "HootOut"."MediaFile" ("Id")
	);

end if; 

CREATE INDEX IF NOT EXISTS ix_chat_message_attachment_message
ON "HootOut"."ChatMessageAttachment" ("MessageId")
WHERE "Deleted" = false;

CREATE INDEX IF NOT EXISTS ix_chat_message_attachment_media_file
ON "HootOut"."ChatMessageAttachment" ("MediaFileId")
WHERE "Deleted" = false;

end $$;
