DO $$
BEGIN

if NOT EXISTS  (
SELECT  schema_name
FROM	information_schema.schemata
WHERE	schema_name = 'HootOut' )
then 
	CREATE SCHEMA "HootOut";
end if;

end $$; 

-- CHAT MESSAGE TABLE ---------------------------------------------------------------
----------------------------------------------------------------------------------

DO $$
BEGIN
if NOT EXISTS (SELECT * FROM INFORMATION_SCHEMA.TABLES
				WHERE TABLE_SCHEMA = 'HootOut'
				AND   TABLE_NAME = 'ChatMessage')
then
	CREATE TABLE "HootOut"."ChatMessage"(
		"Id" UUID NOT NULL PRIMARY KEY ,
		"AuthorId" UUID NOT NULL,
		"ChatChannelId" UUID NOT NULL,
		"Content" TEXT NOT NULL,
		"Deleted" boolean NOT NULL DEFAULT false,
		"CreatedAt" timestamptz  NOT NULL,
		"ModifiedAt" timestamptz NOT NULL,
		CONSTRAINT fk_chat_message_author FOREIGN KEY ("AuthorId") REFERENCES "HootOut"."UserInfo" ("Id"),
		CONSTRAINT fk_chat_message_channel FOREIGN KEY ("ChatChannelId") REFERENCES "HootOut"."ChatChannel" ("Id")
	); 
 
end if;

CREATE INDEX IF NOT EXISTS ix_chat_message_channel_created
ON "HootOut"."ChatMessage" ("ChatChannelId", "CreatedAt" DESC, "Id" DESC)
WHERE "Deleted" = false; 

CREATE INDEX IF NOT EXISTS ix_chat_message_author
ON "HootOut"."ChatMessage" ("AuthorId"); 

end $$;
