DO $$
BEGIN

if NOT EXISTS  (
SELECT  schema_name
FROM	information_schema.schemata
WHERE	schema_name = 'HootOut' )
then 
	CREATE SCHEMA "HootOut";
end if;

end $$; 

-- CHAT SERVER TABLE ---------------------------------------------------------------
----------------------------------------------------------------------------------

DO $$ 
BEGIN
if NOT EXISTS (SELECT * FROM INFORMATION_SCHEMA.TABLES
				WHERE TABLE_SCHEMA = 'HootOut'
				AND   TABLE_NAME = 'ChatServer')
then
	CREATE TABLE "HootOut"."ChatServer"(
		"Id" UUID NOT NULL PRIMARY KEY , 
		"OwnerId" UUID NOT NULL, 
		"Name" TEXT NOT NULL,
		"Description" TEXT,
		"Deleted" boolean NOT NULL DEFAULT false,
		"CreatedAt" timestamptz  NOT NULL,
		"ModifiedAt" timestamptz NOT NULL
	);

end if;  

if NOT EXISTS (SELECT * FROM INFORMATION_SCHEMA.TABLES
				WHERE TABLE_SCHEMA = 'HootOut'
				AND   TABLE_NAME = 'ServerInvitation')
then
	CREATE TABLE "HootOut"."ServerInvitation"(
		"Id" UUID NOT NULL PRIMARY KEY,
		"ChatServerId" UUID NOT NULL REFERENCES "HootOut"."ChatServer"("Id"),
		"InvitedBy" UUID NOT NULL REFERENCES "HootOut"."UserInfo"("Id"),
		"UserId" UUID NOT NULL REFERENCES "HootOut"."UserInfo"("Id"),
		"Status" smallint NOT NULL DEFAULT 0 CHECK ("Status" IN (0, 10, 20, 30)),
		"CreatedAt" timestamptz NOT NULL,
		"ModifiedAt" timestamptz NOT NULL
	);
end if;

CREATE UNIQUE INDEX IF NOT EXISTS ux_server_invitation_pending
ON "HootOut"."ServerInvitation" ("ChatServerId", "UserId")
WHERE "Status" = 0;

CREATE INDEX IF NOT EXISTS ix_server_invitation_invitee
ON "HootOut"."ServerInvitation" ("UserId")
WHERE "Status" = 0;

 

if NOT EXISTS (SELECT * FROM INFORMATION_SCHEMA.TABLES
				WHERE TABLE_SCHEMA = 'HootOut'
				AND   TABLE_NAME = 'ServerMember')
then
	CREATE TABLE "HootOut"."ServerMember"(
		"Id" UUID NOT NULL PRIMARY KEY , 
		"ChatServerId" UUID NOT NULL REFERENCES "HootOut"."ChatServer"("Id"),
		"UserId" UUID NOT NULL REFERENCES "HootOut"."UserInfo"("Id"), 
		"InvitationId" UUID REFERENCES "HootOut"."ServerInvitation"("Id"),
		"Deleted" boolean NOT NULL DEFAULT false,
		"CreatedAt" timestamptz  NOT NULL,
		"ModifiedAt" timestamptz NOT NULL
	);

end if; 

CREATE UNIQUE INDEX IF NOT EXISTS ux_server_member_active
ON "HootOut"."ServerMember" ("ChatServerId", "UserId")
WHERE "Deleted" = FALSE;

CREATE INDEX IF NOT EXISTS ix_server_member_server_createdAt
ON "HootOut"."ServerMember" ("ChatServerId", "CreatedAt" DESC, "Id" DESC);

end $$;
